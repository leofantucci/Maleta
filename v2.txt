void (*reiniciarSoftware)(void) = 0;

void salvarSenhaPontos(){
  if(codigo_inserido == ""){
    if(modoJogoAtual == SABOTAGEM){
      telaErroVazio(MENU_SENHA_PONTOS);
    }
    if(modoJogoAtual == DOMINACAO){
      telaErroVazio(MENU_TEMPO);
    }
  } else{
    if(modoJogoAtual == SABOTAGEM){
      senha = codigo_inserido;
      mudarTela(VISUALIZAR_SENHA_1);
    }
    if(modoJogoAtual == DOMINACAO){
      maxPts = codigo_inserido.toInt();
      mudarTela(VISUALIZAR_SENHA_2);
    } 
  }
}

void telaErroVazio(Tela antiga){
  lcd.clear();
  lcd.setCursor(0,0);
  lcd.print("Digite um valor!");
  delay(1000);
  mudarTela(antiga);
}

void limparSenha(){
  codigo_inserido = "";
  lcd.setCursor(0,1);
  lcd.print("                ");
}

void salvarTempo(){
  if(tempo_inserido == ""){
    telaErroVazio(MENU_TEMPO);
  } else {
    switch(modoJogoAtual){
      case SABOTAGEM:
        tempoExplosao = tempo_inserido;
        mudarTela(VISUALIZAR_TEMPO_1);
        break;

      case DOMINACAO:
        tempoLimite = tempo_inserido;
        if(duracaoAtual == PERSONALIZADA){
          maxPts = tempoLimite.toInt() / 2;
        }
        mudarTela(VISUALIZAR_TEMPO_1);
        break;
    }
  }
}

void limparTempo(){
  tempo_inserido = "";
  lcd.setCursor(0,1);
  lcd.print("                ");
}

int verificarCartao(byte *uidLido, byte tamanho) {
  for (int i = 0; i < NUM_CARTOES; i++) {
    bool igual = true;
    for (int j = 0; j < tamanho; j++) {
      if (uidLido[j] != cartoes[i].uid[j]) {
        igual = false;
        break;
      }
    }
    if (igual) {
      return i;
    }
  }
  return -1;
}

void scrollTexto(ScrollLCD &scroll, int linha, unsigned long intervalo) {
    int tamOriginal = strlen(scroll.texto);
    int tamTotal = tamOriginal + 16; 

    if (millis() - scroll.ultimoScroll >= intervalo) {
        scroll.ultimoScroll = millis();
        lcd.setCursor(0, linha);

        for (int i = 0; i < 16; i++) {
            int indice = (scroll.posicao + i) % tamTotal;
            if (indice < tamOriginal) {
                lcd.print(scroll.texto[indice]);
            } else {
                lcd.print(' ');
            }
        }
        scroll.posicao = (scroll.posicao + 1) % tamTotal;
    }
}

bool apertouBotao(int botao) {
  if (digitalRead(botao) == LOW) {
    while (digitalRead(botao) == LOW) {
      delay(10); 
    }
    return true; 
  }
  return false;
}

void mudarTela(Tela novaTela) {
  telaAtual = novaTela;
  atualizouTela = false;
  
  // Reseta os índices de rolagem dos menus
  menu.posicao = 0;
  inicio.posicao = 0;
  duracaoSabotagem.posicao = 0;
  duracaoDominacao.posicao = 0;
  menuSabotagem.posicao = 0;
  menuDominacao.posicao = 0;
  tempoSabotagem.posicao = 0;
  tempoDominacao.posicao = 0;
  finalAtacantes.posicao = 0;
  finalDefensores.posicao = 0;

  // Reseta estado do jogo quando retorna ao inicio
  if (novaTela == INICIO) {
    pontosEq1 = 0;
    pontosEq2 = 0;
    jaFoi = 0;
    EquipeAtiva = 0;
  }
}

void mudarModoJogo(ModoJogo novoModo) {
  modoJogoAtual = novoModo;
  atualizouModoJogo = false;
}

void mudarModoArmar(ModoArmar novoModo) {
  modoArmarAtual = novoModo;
  atualizouModoArmar = false;
}

void mudarDuracao(Duracao novaDuracao) {
  duracaoAtual = novaDuracao;
  atualizouDuracao = false;
}

void barraCarregamento(int tempo) {
  lcd.createChar(0, bloco);
  unsigned long inicioTimer = millis();
  int posicao = 0;

  lcd.setCursor(0, 1);
  lcd.print("                ");

  while (millis() - inicioTimer < tempo) {
    lcd.setCursor(posicao, 1);
    lcd.write(byte(0));
    delay(250);
    posicao++;

    if (posicao >= 16) {
      posicao = 0;
      lcd.setCursor(0, 1);
      lcd.print("                ");
    }
  }
}

void telaErro(Tela antiga){
  lcd.clear();
  lcd.setCursor(6,0);
  lcd.print("ERRO");
  delay(1000);
  mudarTela(antiga);
}

void telaInicio() {
  ultimaTela = INICIO;
  if (!atualizouTela) {
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Bem-Vindo!");
    atualizouTela = true;
  }

  scrollTexto(inicio, 1, 150);
  
  char tecla = teclado.getKey();
  if (tecla == '1') {
    mudarTela(MODO_JOGO);
  }
}

void telaModoJogo() {
  ultimaTela = INICIO;
  if (!atualizouTela) {
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Escolha o modo:");
    atualizouTela = true;
  }

  scrollTexto(menu, 1, 150);

  char tecla = teclado.getKey();
  if(tecla == 'B'){
    mudarTela(ultimaTela);
  }
  if (tecla == '1') {
    mudarModoJogo(SABOTAGEM);
    mudarTela(MODO_ARMAR);
  }
  if (tecla == '2') {
    mudarModoJogo(DOMINACAO);
    mudarTela(MODO_ARMAR);
  }
}

void telaModoArmar() {
  ultimaTela = MODO_JOGO;
  if (!atualizouTela) {
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Modo de armar:");
    atualizouTela = true;
  }
  
  char tecla = teclado.getKey();
  if(tecla == 'B'){
    mudarTela(ultimaTela);
  }

  switch(modoJogoAtual){
    case SABOTAGEM:
      scrollTexto(menuSabotagem, 1, 150);

      if (tecla == '1') {
        mudarModoArmar(BOTAO);
        mudarTela(DURACAO);
      }
      if (tecla == '2') {
        mudarModoArmar(CARTAO);
        mudarTela(DURACAO);
      }
      if (tecla == '3') {
        mudarModoArmar(SENHA);      
        mudarTela(MENU_SENHA_PONTOS);
      }
      break;
      
    case DOMINACAO:
      scrollTexto(menuDominacao, 1, 150);
    
      if (tecla == '1') {
        mudarModoArmar(BOTAO);
        mudarTela(DURACAO);
      }
      if (tecla == '2') {
        mudarModoArmar(CARTAO);
        mudarTela(DURACAO);
      }
      break;
  }
}

void telaDuracao(){
  ultimaTela = MODO_ARMAR;
  if (!atualizouTela) {
    lcd.clear();
    lcd.setCursor(0, 0);
    if(modoJogoAtual == SABOTAGEM){
      lcd.print("Tempo apos armar");
    } else if(modoJogoAtual == DOMINACAO){
      lcd.print("Limite de tempo");
    }
    atualizouTela = true;
  }  

  if(modoJogoAtual == SABOTAGEM){
    scrollTexto(duracaoSabotagem, 1, 150);
  }
  if(modoJogoAtual == DOMINACAO){
    scrollTexto(duracaoDominacao, 1, 150);
  }
  
  char tecla = teclado.getKey();

  if((tecla >= '1' && tecla <= '4') || tecla == 'B'){
    switch(tecla){
      case 'B':
        mudarTela(ultimaTela);
        break;
      case '1':
        mudarDuracao(CQB);
        if(modoJogoAtual == SABOTAGEM) tempoExplosao = "50";
        if(modoJogoAtual == DOMINACAO) { tempoLimite = "600"; maxPts = 300; }
        mudarTela(JOGANDO);
        break;
      case '2':
        mudarDuracao(PADRAO);
        if(modoJogoAtual == SABOTAGEM) tempoExplosao = "90";
        if(modoJogoAtual == DOMINACAO) { tempoLimite = "1800"; maxPts = 900; }
        mudarTela(JOGANDO);
        break;
      case '3':
        mudarDuracao(CAMPO_ABERTO);
        if(modoJogoAtual == SABOTAGEM) tempoExplosao = "260";
        if(modoJogoAtual == DOMINACAO) { tempoLimite = "3600"; maxPts = 1800; }
        mudarTela(JOGANDO);
        break;
      case '4':
        mudarDuracao(PERSONALIZADA);
        mudarTela(MENU_TEMPO);
        break;
    }
  }
}

void telaMenuSenhaPontos() {
  ultimaTela = MODO_ARMAR;
  if (!atualizouTela) {
    codigo_inserido = "";
    lcd.clear();
    lcd.setCursor(0, 0);
    if(modoJogoAtual == SABOTAGEM){ 
      lcd.print("Defina a senha:");
    }
    if(modoJogoAtual == DOMINACAO){ 
      lcd.print("Def. pts. max.:");
    }
    atualizouTela = true;
  }

  char tecla = teclado.getKey();
  if(tecla >= '0' && tecla <= '9'){
    codigo_inserido += tecla;
    lcd.setCursor(0,1);
    lcd.print("                ");
    lcd.setCursor(0,1);
    lcd.print(codigo_inserido);
  }
  if(tecla == 'D' || apertouBotao(BTN_VERMELHO)){
    limparSenha();
  }
  if(tecla == '#' || apertouBotao(BTN_VERDE)){
    salvarSenhaPontos();
  }
  if(tecla == 'B'){
    mudarTela(ultimaTela);
  }
}

  }

  }
}

void telaJogando(){
  if (!atualizouTela) {
    int pos = 0;
    String texto = "";
    lcd.clear();
    lcd.setCursor(0,0);

    switch(modoJogoAtual){
      case SABOTAGEM:
        texto = "ARME A BOMBA";
        lcd.print(texto);
        pos = texto.length() + 1;
        break;

      case DOMINACAO:
        texto = "DOMINE A AREA";
        lcd.print(texto);
        pos = texto.length() + 1;
        break;
    }

    lcd.setCursor(pos,0);
    switch(modoArmarAtual){
      case SENHA:
        codigo_inserido = "";
        lcd.print("- 3");
        break;
      case CARTAO:
        lcd.print("- 2");
        break;
      case BOTAO:
        lcd.print("- 1");
        break;
    }
    atualizouTela = true;
  }

  char tecla = teclado.getKey();

  switch(modoJogoAtual){
    case SABOTAGEM:
      switch(modoArmarAtual){
        case BOTAO:
          if(apertouBotao(BTN_VERDE)){
            mudarTela(ARMANDO);
          }
          if(apertouBotao(BTN_VERMELHO) || tecla) {
            telaErro(JOGANDO);
          }
          break;

        case SENHA:
          if(tecla >= '0' && tecla <= '9'){
            codigo_inserido += tecla;
            lcd.setCursor(0,1);
            lcd.print("                ");
            lcd.setCursor(0,1);
            lcd.print(codigo_inserido);
          }
          if(tecla == 'A' || tecla == '#' || apertouBotao(BTN_VERDE)){
            if(codigo_inserido == senha){
              mudarTela(ARMANDO);
            } else{
              telaErro(JOGANDO);
            }
          }
          break;

        case CARTAO:
          if (rfid.PICC_IsNewCardPresent() && rfid.PICC_ReadCardSerial()) {
            int indice = verificarCartao(rfid.uid.uidByte, rfid.uid.size);
            if (indice != -1 && cartoes[indice].nome == "Verde") {
              mudarTela(ARMANDO);
            } else {
              telaErro(JOGANDO);
            }
            rfid.PICC_HaltA();
            rfid.PCD_StopCrypto1();
          }
          break;
      }
      break;
    
    case DOMINACAO:
      switch(modoArmarAtual){
        case BOTAO:
          if(apertouBotao(BTN_VERDE) || tecla == '1'){
            mudarTela(DOMINANDO_1);
          }
          if(apertouBotao(BTN_VERMELHO) || tecla == '2') {
            mudarTela(DOMINANDO_2);
          }
          break;

        case CARTAO:
          if (rfid.PICC_IsNewCardPresent() && rfid.PICC_ReadCardSerial()) {
            int indice = verificarCartao(rfid.uid.uidByte, rfid.uid.size);
            if (indice != -1) {
              if(cartoes[indice].nome == "Verde"){
                mudarTela(DOMINANDO_1);
              } else if(cartoes[indice].nome == "Vermelho"){
                mudarTela(DOMINANDO_2);
              }
            } else {
              telaErro(JOGANDO);
            }
            rfid.PICC_HaltA();
            rfid.PCD_StopCrypto1();
          }
          break;
      }
      break;
  }
}

void telaDominando(int key){
  if(!atualizouTela){
    lcd.clear();
    lcd.setCursor(0,0);
    EquipeAtiva = key;

    lcd.print("DOMINANDO A AREA");
    if((tempoLimite.toInt()/10) <= 80){
      barraCarregamento(4000);
    } else{
      barraCarregamento(8000);
    }
    
    atualizouTela = true;
  }

  lcd.clear();
  mudarTela(DOMINADA);
}

void telaDominada(){
  if(!atualizouTela){
    lcd.clear();
    lcd.createChar(0, setaEsquerda);

    pos1 = (EquipeAtiva == 1) ? 2 : 0;
    pos2 = (EquipeAtiva == 2) ? 2 : 0;

    if(EquipeAtiva == 1){
      lcd.setCursor(0, 0);
      lcd.write(byte(0));
    } else if(EquipeAtiva == 2){
      lcd.setCursor(0, 1);
      lcd.write(byte(0));
    }

    lcd.setCursor(pos1, 0);
    lcd.print("EQ. 1: ");
    pos1 += 7;

    lcd.setCursor(pos2, 1);
    lcd.print("EQ. 2: ");
    pos2 += 7;

    if(jaFoi == 0){
      tempoInicioDominacao = millis();
      ultimoPontoContado = millis();
      jaFoi = 1;
    }
    atualizouTela = true;
  }

  // --- ACÚMULO DE PONTOS A CADA 1 SEGUNDO ---
  if(millis() - ultimoPontoContado >= freq){
    ultimoPontoContado = millis();
    if(EquipeAtiva == 1) pontosEq1++;
    if(EquipeAtiva == 2) pontosEq2++;
  }

  // Atualiza pontos na tela
  lcd.setCursor(pos1, 0);
  lcd.print(pontosEq1);
  lcd.print("   ");

  lcd.setCursor(pos2, 1);
  lcd.print(pontosEq2);
  lcd.print("   ");

  char tecla = teclado.getKey();
  
  // Verificação de Tempo Limite do Jogo
  unsigned long tempoDecorridoSeg = (millis() - tempoInicioDominacao) / 1000;
  if(tempoDecorridoSeg >= tempoLimite.toInt()){
    mudarTela(FINAL);
    return;
  }

  // Verificação de Vitória por Pontos Máximos
  if(pontosEq1 >= maxPts || pontosEq2 >= maxPts){
    mudarTela(FINAL);
    return;
  }

  // Troca de Equipe Ativa via Teclado ou Botões
  switch(modoArmarAtual){
    case CARTAO:
      if (rfid.PICC_IsNewCardPresent() && rfid.PICC_ReadCardSerial()) {
        int indice = verificarCartao(rfid.uid.uidByte, rfid.uid.size);
        if (indice != -1) {
          if(cartoes[indice].nome == "Verde" && EquipeAtiva != 1){
            EquipeAtiva = 1;
            mudarTela(DOMINADA);
          } else if(cartoes[indice].nome == "Vermelho" && EquipeAtiva != 2){
            EquipeAtiva = 2;
            mudarTela(DOMINADA);
          }
        } else {
          telaErro(DOMINADA);
        }
        rfid.PICC_HaltA();
        rfid.PCD_StopCrypto1();
      }
      break;

    case BOTAO:
      if((tecla == 'A' || tecla == '1' || apertouBotao(BTN_VERDE)) && EquipeAtiva != 1){
        EquipeAtiva = 1;
        mudarTela(DOMINADA);
      }
      if((tecla == 'B' || tecla == '2' || apertouBotao(BTN_VERMELHO)) && EquipeAtiva != 2){
        EquipeAtiva = 2;
        mudarTela(DOMINADA);
      }
      break;
  }
}

void telaArmando(){
  if(!atualizouTela){
    lcd.clear();
    lcd.setCursor(1,0);
    lcd.print("ARMANDO  BOMBA");

    if((tempoExplosao.toInt()/10) >= 8){
      barraCarregamento(8000);
    } else{
      barraCarregamento(4000);
    }
    atualizouTela = true;
  }

  lcd.clear();
  lcd.setCursor(2,0);
  lcd.print("BOMBA ARMADA");

  delay(1000);
  inicioExplosao = millis();
  mudarTela(ARMADA);
}

void telaArmada(){
  if(!atualizouTela){
    lcd.clear();
    lcd.setCursor(2,0);
    lcd.print("BOMBA ARMADA");
    atualizouTela = true;
  }

  int tempoRestante = tempoExplosao.toInt() - ((millis() - inicioExplosao) / 1000);
  if(tempoRestante < 0) tempoRestante = 0;

  char bufferTempo[17];
  if (tempoRestante >= 60) {
    int minutos = tempoRestante / 60;
    int segundos = tempoRestante % 60;
    snprintf(bufferTempo, sizeof(bufferTempo), "Tempo: %dmin%ds", minutos, segundos);
  } else {
    snprintf(bufferTempo, sizeof(bufferTempo), "Tempo: %dseg.", tempoRestante);
  }

  lcd.setCursor(0, 1);
  lcd.print(bufferTempo);
  
  int tamTexto = strlen(bufferTempo);
  for (int i = tamTexto; i < 16; i++) {
    lcd.print(" ");
  }

  if(tempoRestante <= 0){
    mudarTela(FINAL_ATAQUE);
    return;
  }

  bool querDesarmar = false;

  if (modoArmarAtual == SENHA) {
    teclado.getKeys();
    for (int i = 0; i < LIST_MAX; i++) {
      if ((teclado.key[i].kchar == 'D' && teclado.key[i].kstate == HOLD) || digitalRead(BTN_VERMELHO) == LOW) {
        querDesarmar = true;
        break;
      }
    }
  } 
  else if (modoArmarAtual == BOTAO) {
    if (digitalRead(BTN_VERMELHO) == LOW) {
      querDesarmar = true;
    }
  } 
  else if (modoArmarAtual == CARTAO) {
    if (rfid.PICC_IsNewCardPresent() && rfid.PICC_ReadCardSerial()) {
      int indice = verificarCartao(rfid.uid.uidByte, rfid.uid.size);
      if (indice != -1 && cartoes[indice].nome == "Vermelho") {
        querDesarmar = true;
      } else{
        telaErro(ARMADA);
      }
      rfid.PICC_HaltA();
      rfid.PCD_StopCrypto1();
    }
  }

  if (querDesarmar) {
    progressoAnimacao = 0;
    ultimoFrameAnimacao = millis();
    mudarTela(DESARMANDO);
  }
}

void telaFinal(int key){
  if(modoJogoAtual == DOMINACAO){
    if(!atualizouTela){
      lcd.clear();
      int equipeVencedora = 0;
      int ptsVencedor = 0;

      if(pontosEq1 > pontosEq2){
        equipeVencedora = 1;
        ptsVencedor = pontosEq1;
      } else if(pontosEq2 > pontosEq1){
        equipeVencedora = 2;
        ptsVencedor = pontosEq2;
      } else {
        equipeVencedora = 3;
        ptsVencedor = pontosEq1;
      }

      if(equipeVencedora == 3){
        lcd.setCursor(1,0);
        lcd.print("!!! EMPATE !!!");
      } else {
        lcd.setCursor(0,0);
        lcd.print("VENCE A EQUIPE ");
        lcd.print(equipeVencedora);
      }

      lcd.setCursor(0,1);
      lcd.print("COM ");
      lcd.print(ptsVencedor);
      lcd.print(" PTS.");
      atualizouTela = true;
    }
  }

  if(modoJogoAtual == SABOTAGEM){
    if(!atualizouTela){
      lcd.clear();
      lcd.setCursor(1,0);
      lcd.print("FINAL DE JOGO!");
      atualizouTela = true;
    }
    switch(key){
      case 0:
        scrollTexto(finalDefensores, 1, 150);
        break;
      case 1:
        scrollTexto(finalAtacantes, 1, 150);
        break;
    }
  }

  char tecla = teclado.getKey();
  if(tecla){
    lcd.clear();
    lcd.print("REINICIANDO...");
    delay(1500);
    reiniciarSoftware();
  }
}

void telaDesarmando() {
  int tempoRestante = tempoExplosao.toInt() - ((millis() - inicioExplosao) / 1000);
  if (tempoRestante < 0) tempoRestante = 0;

  if (tempoRestante <= 0) {
    mudarTela(FINAL_ATAQUE);
    return;
  }

  if (!atualizouTela) {
    lcd.clear();
    lcd.createChar(0, bloco);
    atualizouTela = true;
  }

  char bufferTopo[17];
  if (tempoRestante >= 60) {
    int minutos = tempoRestante / 60;
    int segundos = tempoRestante % 60;
    snprintf(bufferTopo, sizeof(bufferTopo), "DESARM. %dmin%ds", minutos, segundos);
  } else {
    snprintf(bufferTopo, sizeof(bufferTopo), "DESARM. %dseg.", tempoRestante);
  }

  lcd.setCursor(0, 0);
  lcd.print(bufferTopo);
  
  int tamTexto = strlen(bufferTopo);
  for (int i = tamTexto; i < 16; i++) {
    lcd.print(" ");
  }

  bool mantendoAcao = false;

  if (modoArmarAtual == SENHA) {
    teclado.getKeys();
    bool segurandoD = false;
    for (int i = 0; i < LIST_MAX; i++) {
      if (teclado.key[i].kchar == 'D' && (teclado.key[i].kstate == HOLD || teclado.key[i].kstate == PRESSED)) {
        segurandoD = true;
        break;
      }
    }
    if (segurandoD || digitalRead(BTN_VERMELHO) == LOW) {
      mantendoAcao = true;
    }
  } 
  else if (modoArmarAtual == BOTAO) {
    if (digitalRead(BTN_VERMELHO) == LOW) {
      mantendoAcao = true;
    }
  } 
  else if (modoArmarAtual == CARTAO) {
    if (rfid.PICC_IsNewCardPresent() && rfid.PICC_ReadCardSerial()) {
      int indice = verificarCartao(rfid.uid.uidByte, rfid.uid.size);
      if (indice != -1 && cartoes[indice].nome == "Vermelho") {
        mantendoAcao = true;
      }
      rfid.PICC_HaltA();
      rfid.PCD_StopCrypto1();
    }
  }

  if (!mantendoAcao && modoArmarAtual != CARTAO) {
    mudarTela(ARMADA);
    return;
  }

  if (millis() - ultimoFrameAnimacao >= 500) {
    ultimoFrameAnimacao = millis();

    lcd.setCursor(progressoAnimacao, 1);
    lcd.write(byte(0));
    progressoAnimacao++;

    if (progressoAnimacao >= 17) {
      mudarTela(FINAL_DEFESA);
    }
  }
}

void setup() {
  Serial.begin(115200);

  // LCD
  lcd.init();
  lcd.backlight();
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Sistema");

  // RFID
  SPI.begin();
  rfid.PCD_Init();

  // Botões
  pinMode(BTN_VERDE, INPUT_PULLUP);
  pinMode(BTN_VERMELHO, INPUT_PULLUP);

  Serial.println("Sistema iniciado");
}

void loop() {
  switch (telaAtual) {
    case INICIO:
      telaInicio();
      break;

    case MODO_JOGO:
      telaModoJogo();
      break;

    case DURACAO:
      telaDuracao();
      break;

    case MODO_ARMAR:
      telaModoArmar();
      break;

    case MENU_SENHA_PONTOS:
      telaMenuSenhaPontos();
      break;

    case MENU_TEMPO:
      telaMenuTempo();
      break;

    case VISUALIZAR_SENHA_1:
      telaVisualizarSenha(1);
      break;

    case VISUALIZAR_SENHA_2:
      telaVisualizarSenha(2);
      break;

    case VISUALIZAR_TEMPO_1:
      telaVisualizarTempo(1);
      break;

    case JOGANDO:
      telaJogando();
      break;

    case ARMANDO:
      telaArmando();
      break;

    case ARMADA:
      telaArmada();
      break;

    case DESARMANDO:
      telaDesarmando();
      break;
    
    case FINAL:
      telaFinal(9);
      break;

    case FINAL_ATAQUE:
      telaFinal(1);
      break;

    case FINAL_DEFESA:
      telaFinal(0);
      break;

    case DOMINANDO_1:
      telaDominando(1);
      break;

    case DOMINANDO_2:
      telaDominando(2);
      break;

    case DOMINADA:
      telaDominada();
      break;

    default:
      mudarTela(INICIO);
      break;
  }
}
